CREATE TABLE Insurance_Policies (
    policy_id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT,
    company_id INT,
    policy_number VARCHAR(50) UNIQUE,
    coverage_amount DECIMAL(10,2),
    start_date DATE,
    end_date DATE,
    FOREIGN KEY (patient_id) REFERENCES Patients(patient_id),
    FOREIGN KEY (company_id) REFERENCES Insurance_Companies(company_id)
);

CREATE TABLE Appointments (
    appointment_id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT,

SELECT d.doctor_name,
       COUNT(a.patient_id) AS total_patients
FROM Doctors d
JOIN Appointments a
ON d.doctor_id = a.doctor_id
GROUP BY d.doctor_name;

SELECT p.first_name,
       p.last_name,
       b.total_amount
FROM Patients p
JOIN Bills b
ON p.patient_id = b.patient_id
ORDER BY b.total_amount DESC

SELECT d.doctor_name,
       COUNT(a.patient_id) AS total_patients
FROM Doctors d
JOIN Appointments a
ON d.doctor_id = a.doctor_id
GROUP BY d.doctor_name
ORDER BY total_patients DESC
LIMIT 1;

SELECT p.first_name,
       p.last_name
FROM Patients p
LEFT JOIN Insurance_Policies ip
ON p.patient_id = ip.patient_id
LEFT JOIN Insurance_Claims ic
ON ip.policy_id = ic.policy_id
WHERE ic.claim_id IS NULL;

SELECT ic.company_name,
       SUM(c.claim_amount) AS total_claim
FROM Insurance_Companies ic
JOIN Insurance_Policies ip
ON ic.company_id = ip.company_id
JOIN Insurance_Claims c
ON ip.policy_id = c.policy_id
GROUP BY ic.company_name
ORDER BY total_claim DESC
LIMIT 1;
